Code Tabs

# Create a vector and compute its sum
v = [1, 2, 3, 4, 5]
println(sum(v))   # 15
# Create a list and compute its sum
v = [1, 2, 3, 4, 5]
print(sum(v))   # 15
# Create a vector and compute its sum
v <- c(1, 2, 3, 4, 5)
print(sum(v))   # 15
